# ADR-0007: 质押与收益分发 sM

## 状态

已采纳

## 背景

StakedUSDeV2 是基于 ERC-4626 标准的不变合约。用户存入 USDe 获得 sUSDe，协议通过 `REWARDER` 角色定期注入 USDe 奖励，sUSDe:USDe 汇率单调递增。奖励在 8 小时内线性 vesting 防止操纵。解质押有可选冷却期（最长 90 天）。

## 决策

### 合约结构

```solidity
contract StakedM is ERC20, ERC4626, AccessControl, ReentrancyGuard {
    using SafeERC20 for IERC20;

    // 角色
    bytes32 public constant REWARDER_ROLE = keccak256("REWARDER_ROLE");
    bytes32 public constant BLACKLISTER_ROLE = keccak256("BLACKLISTER_ROLE");
    bytes32 public constant SOFT_RESTRICTED_STAKER_ROLE = keccak256("SOFT_RESTRICTED_STAKER_ROLE");
    bytes32 public constant FULL_RESTRICTED_STAKER_ROLE = keccak256("FULL_RESTRICTED_STAKER_ROLE");

    // Vesting
    uint256 public constant VESTING_DURATION = 8 hours;
    uint256 private _vestingStart;
    uint256 private _vestingEnd;
    uint256 private _totalRewards;

    // Cooldown
    uint256 public cooldownDuration;
    mapping(address => RedemptionRequest) public redemptionRequests;

    struct RedemptionRequest {
        uint256 shares;
        uint256 requestBlock;
        bool claimed;
    }
}
```

### 质押/解质押

```solidity
// 质押：用户存入 M，获得 sM
function deposit(uint256 assets, address receiver) public override returns (uint256) {
    _checkNotRestricted(receiver);
    return super.deposit(assets, receiver);
}

// 解质押请求（有冷却期）
function requestRedeem(uint256 shares) external {
    if (hasRole(FULL_RESTRICTED_STAKER_ROLE, msg.sender)) revert RestrictedAddress();

    uint256 assets = previewRedeem(shares);
    _burn(msg.sender, shares);
    _transfer(asset(), address(this), assets);

    redemptionRequests[msg.sender] = RedemptionRequest({
        shares: shares,
        requestBlock: block.number,
        claimed: false
    });
    emit RedemptionRequested(msg.sender, shares, block.timestamp);
}

// 提取（冷却期后）
function withdraw(uint256 assets, address receiver, address owner) public override returns (uint256) {
    RedemptionRequest storage req = redemptionRequests[msg.sender];
    if (req.claimed) revert AlreadyClaimed();
    if (block.timestamp < req.requestBlock + cooldownDuration) revert CooldownNotElapsed();

    req.claimed = true;
    IERC20(asset()).safeTransfer(msg.sender, assets);
    emit WithdrawalCompleted(msg.sender, req.shares, assets);
}
```

### 奖励累积

```solidity
// REWARDER 注入奖励
function transferInRewards(uint256 amount) external onlyRole(REWARDER_ROLE) {
    // 合并未 vest 奖励
    if (_totalRewards > 0 && block.timestamp < _vestingEnd) {
        uint256 elapsed = block.timestamp - _vestingStart;
        uint256 remaining = _totalRewards * (VESTING_DURATION - elapsed) / VESTING_DURATION;
        _totalRewards = remaining + amount;
    } else {
        _totalRewards = amount;
    }
    _vestingStart = block.timestamp;
    _vestingEnd = block.timestamp + VESTING_DURATION;

    IERC20(asset()).safeTransferFrom(msg.sender, address(this), amount);
    emit RewardsReceived(amount);
}

// 总资产 = 实际余额 - 未 vest 奖励
function totalAssets() public view override returns (uint256) {
    uint256 unvested = 0;
    if (_totalRewards > 0 && block.timestamp < _vestingEnd) {
        uint256 elapsed = block.timestamp - _vestingStart;
        unvested = _totalRewards * (VESTING_DURATION - elapsed) / VESTING_DURATION;
    }
    return IERC20(asset()).balanceOf(address(this)) - unvested;
}

// sM:M 汇率单调递增（只能正或平）
function convertToAssets(uint256 shares) public view override returns (uint256) {
    uint256 supply = totalSupply();
    if (supply == 0) return shares;
    return shares * totalAssets() / supply;
}
```

### 奖励分发自动化

```solidity
contract StakingRewardsDistributor is AccessControl {
    bytes32 public constant OPERATOR_ROLE = keccak256("OPERATOR_ROLE");

    IStakedM public stakedM;
    IM public m;

    // 每周多次小额分发防操纵
    uint256 public constant DISTRIBUTION_INTERVAL = 1 days;
    uint256 public lastDistribution;

    function distribute(uint256 amount) external onlyRole(OPERATOR_ROLE) {
        m.approve(address(stakedM), amount);
        stakedM.transferInRewards(amount);
        lastDistribution = block.timestamp;
    }
}
```

### 合规限制

```solidity
function _checkNotRestricted(address account) internal view {
    if (hasRole(SOFT_RESTRICTED_STAKER_ROLE, account)) revert SoftRestrictedAddress();
    if (hasRole(FULL_RESTRICTED_STAKER_ROLE, account)) revert RestrictedAddress();
}

// FULL_RESTRICTED 用户的 sM 可被协议接管（类似 Circle USDC）
function redistributeLockedAmount(address from, address to) external onlyRole(BLACKLISTER_ROLE) {
    if (!hasRole(FULL_RESTRICTED_STAKER_ROLE, from)) revert NotRestricted();
    uint256 shares = balanceOf(from);
    _burn(from, shares);
    _mint(to, shares);
    emit LockedAmountRedistributed(from, to, shares);
}
```

## 后果

### 正面
- ERC-4626 标准，与 Aave、Pendle、Morpho 等协议可组合
- 8 小时 vesting 防止闪电贷套利
- sM: M 汇率只增不减（负收益由 Reserve Fund 吸收）

### 负面
- 合规限制（SOFT/FULL_RESTRICTED）引入审查担忧
- 冷却期降低用户体验
- Vesting 计算增加 Gas 开销
